


#include <stdio.h>

#define TAM 5

int main(void) {
    int numeros[TAM];
    int i, j, temp;

    printf("=== Ordenacao de Vetores ===\n");

    for (i = 0; i < TAM; i++) {
        printf("Digite o %d numero: ", i + 1);
        if (scanf("%d", &numeros[i]) != 1) {
            printf("Entrada invalida!\n");
            return 1;
        }
    }

    for (i = 0; i < TAM - 1; i++) {
        for (j = 0; j < TAM - 1 - i; j++) {
            if (numeros[j] > numeros[j + 1]) {
                temp = numeros[j];
                numeros[j] = numeros[j + 1];
                numeros[j + 1] = temp;
            }
        }
    }

    printf("Numeros em ordem crescente:\n");
    for (i = 0; i < TAM; i++) {
        printf("%d ", numeros[i]);
    }

    printf("\n");
    return 0;
}
